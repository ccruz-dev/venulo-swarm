/**
 * Validates all required and optional environment variables on server boot.
 * Call this as the very first thing in server/index.js.
 */
function validateEnv() {
  const required = {
    // The Prisma schema/provider is PostgreSQL â€” a file: (SQLite) URL will make
    // every Prisma call throw at runtime even though the server boots.
    DATABASE_URL: { validate: (v) => v.startsWith('postgresql://') || v.startsWith('postgres://'), error: 'Must be a PostgreSQL connection string. The schema provider is postgresql â€” SQLite file: URLs no longer work; point dev at a local or Railway Postgres.' },
    JWT_SECRET: { validate: (v) => v.length >= 64, error: 'Must be at least 64 characters' },
  };

  // Production-only required variables
  const productionRequired = {
    ENCRYPTION_KEY: { validate: (v) => /^[0-9a-f]{64}$/i.test(v), error: 'Must be exactly 64 hex characters' },
    STRIPE_SECRET_KEY: { validate: (v) => v.startsWith('sk_'), error: 'Must start with sk_' },
    STRIPE_WEBHOOK_SECRET: { validate: (v) => v.startsWith('whsec_'), error: 'Must start with whsec_' },
    STRIPE_BASIC_PRICE_ID: { validate: (v) => v.startsWith('price_'), error: 'Must start with price_' },
    STRIPE_PRO_PRICE_ID: { validate: (v) => v.startsWith('price_'), error: 'Must start with price_' },
    STRIPE_ENTERPRISE_PRICE_ID: { validate: (v) => v.startsWith('price_'), error: 'Must start with price_' },
    STRIPE_ONBOARDING_PRICE_ID: { validate: (v) => v.startsWith('price_'), error: 'Must start with price_' },
    RESEND_API_KEY: { validate: (v) => v.startsWith('re_'), error: 'Must start with re_' },
    FRONTEND_URL: { validate: (v) => v.startsWith('http'), error: 'Must be a valid URL' },
    INTERNAL_API_KEY: { validate: (v) => v.length >= 32, error: 'Must be at least 32 characters' },
    // Admin tokens are minted as jwt.sign({ role: 'admin' }, ADMIN_JWT_SECRET).
    // A guessable value here means anyone can forge an admin session, so this is
    // length-checked like JWT_SECRET. (Found in the wild set to a 9-character
    // dictionary word + digits, with nothing validating it.)
    ADMIN_JWT_SECRET: { validate: (v) => v.length >= 32, error: 'Must be at least 32 characters â€” it signs admin sessions' },
  };

  const optional = {
    ENABLE_MARKETING_EMAILS: { warn: 'Automated marketing is disabled by default in production during the pilot.' },
    ENABLE_FREE_REPORTS: { warn: 'Anonymous free-report processing is disabled by default in production during the pilot.' },
    ENABLE_POS_INTEGRATIONS: { warn: 'POS connections and automatic sync are disabled by default in production during the pilot.' },
    STORAGE_ROOT: { warn: 'Persistent file storage is not configured. Mount a Railway volume and set STORAGE_ROOT to its absolute mount path before accepting customer uploads.' },
    REDIS_URL: { warn: 'Redis not configured â€” job queue will use in-memory fallback. Not recommended for production.' },
    SENTRY_DSN: { warn: 'Sentry not configured â€” errors will not be tracked.' },
    ANTHROPIC_API_KEY: { warn: 'Anthropic API key not configured â€” AI insights will be disabled.' },
    DIRECT_DATABASE_URL: { warn: 'Direct database URL not configured â€” migrations must use the main DATABASE_URL.' },
    GOOGLE_CLIENT_ID: { warn: 'Google OAuth not configured â€” "Sign in with Google" will be hidden.' },
    POSTHOG_API_KEY: { warn: 'PostHog not configured â€” server-side product analytics disabled (no-op).' },
    // Both are needed together; one alone silently does nothing, which is the
    // failure mode worth warning about since it bills pilots at full price.
    PILOT_ACCESS_CODE: { warn: 'Pilot access code not set â€” the pilot discount cannot be claimed at signup.' },
    STRIPE_PILOT_COUPON_ID: { warn: 'Pilot coupon not set â€” the pilot discount cannot be applied.' },
    HEALTH_DETAIL_TOKEN: { warn: 'Health detail token not set â€” /health returns a bare status to everyone (safe default). Set it to read the detailed checks via the x-health-token header.' },
  };

  const errors = [];
  const warnings = [];
  const isProd = process.env.NODE_ENV === 'production';
  for (const name of ['ENABLE_MARKETING_EMAILS', 'ENABLE_FREE_REPORTS', 'ENABLE_POS_INTEGRATIONS']) {
    if (process.env[name] !== undefined && !['true', 'false'].includes(process.env[name])) errors.push(`  - ${name}: Must be true or false`);
  }
  if (process.env.STORAGE_ROOT && !require('path').isAbsolute(process.env.STORAGE_ROOT)) errors.push('  - STORAGE_ROOT must be an absolute path');

  for (const [key, config] of Object.entries(required)) {
    const val = process.env[key];
    if (!val) {
      errors.push(`  - ${key}: Missing (${config.error})`);
    } else if (!config.validate(val)) {
      errors.push(`  - ${key}: ${config.error}`);
    }
  }

  // Check production-only required vars
  if (isProd) {
    for (const [key, config] of Object.entries(productionRequired)) {
      const val = process.env[key];
      if (!val) {
        errors.push(`  - ${key}: Missing (${config.error})`);
      } else if (!config.validate(val)) {
        errors.push(`  - ${key}: ${config.error}`);
      }
    }
  }

  // Check optional vars
  for (const [key, config] of Object.entries(optional)) {
    if (!process.env[key]) {
      warnings.push(`  - ${key}: ${config.warn}`);
    }
  }

  // Production-mode sanity warnings â€” things that boot fine but break customers
  if (isProd) {
    if (!process.env.FROM_EMAIL) {
      warnings.push('  - FROM_EMAIL: Not set â€” emails will send from Resend\'s sandbox address (onboarding@resend.dev), which can ONLY deliver to the Resend account owner. Customer emails WILL fail. Verify a domain at resend.com/domains and set FROM_EMAIL.');
    }
    // Half-configured pilots are worse than none: a pilot types the code, the
    // signup succeeds, and they are billed the full rate with no error anywhere.
    const hasPilotCode = Boolean(process.env.PILOT_ACCESS_CODE);
    const hasPilotCoupon = Boolean(process.env.STRIPE_PILOT_COUPON_ID);
    if (hasPilotCode !== hasPilotCoupon) {
      warnings.push(`  - PILOT DISCOUNT HALF-CONFIGURED: ${hasPilotCode ? 'PILOT_ACCESS_CODE is set but STRIPE_PILOT_COUPON_ID is not' : 'STRIPE_PILOT_COUPON_ID is set but PILOT_ACCESS_CODE is not'}. Pilots will be charged FULL PRICE with no error shown. Set both or neither.`);
    }
    if (process.env.STRIPE_SECRET_KEY?.startsWith('sk_test_')) {
      warnings.push('  - STRIPE_SECRET_KEY: Using a TEST-mode key in production â€” checkouts will be sandbox transactions, no real charges. Switch to sk_live_ keys, live price IDs, and a live webhook endpoint secret when going live.');
    }
  }

  // Log warnings
  if (warnings.length > 0) {
    logger.warn('Environment warnings:');
    warnings.forEach(w => logger.warn(w));
  }

  // Fatal on errors
  if (errors.length > 0) {
    logger.fatal('FATAL: Environment configuration errors:');
    errors.forEach(e => logger.fatal(e));
    logger.fatal('Server cannot start. Fix these environment variables and restart.');
    process.exit(1);
  }
}

# Venulo â€” Project Documentation

## Architecture

**Monorepo** with two directories:
- `client/` â€” React 18 + Vite 5 + TailwindCSS 3.4 + shadcn/ui
- `server/` â€” Express 4 + Prisma ORM + PostgreSQL

**Production deployment:** Single Express server serves both the API (`/api/*`) and the built Vite client (static files from `client/dist/`). Deployed on Railway.

**Database:** PostgreSQL (Railway). Prisma ORM with soft-delete middleware.

**Auth:** JWT in httpOnly cookies with refresh token rotation. CSRF double-submit cookies. Account lockout after 10 failed attempts. Optional TOTP 2FA.

**Key services:** BullMQ job queue (Redis), Stripe subscriptions, Anthropic Claude API (AI insights, optional), nodemailer (SMTP email), PDFKit (report generation).

## Destructive Operations Against Production

Written after 2026-09-14, when an authorized instruction to "delete every
account except one" silently also destroyed 10 locations, 4 uploads, 4 reports
and 15,576 transactions. The deletion was legitimate. The failure was that the
cascade was never counted or disclosed, so the result looked like an
unexplained wipe and cost hours of incident investigation.

**Before ANY destructive operation on production data:**

1. **Count the dependent rows first and state them.** The schema declares 30
   `onDelete: Cascade` relations. `User` cascades to `Location`, which cascades
   onward to `Upload`, `Report`, `Transaction` and others, and `Upload` and
   `Report` cascade further still. Deleting N users is never "deleting N rows."
2. **Say the full blast radius before running it**, not after. "This removes 9
   users, 9 locations, 4 uploads, 4 reports and 15,576 transactions" â€” then act.
3. **Take a verified backup first** (`node scripts/backup-encrypted.js`) and do
   not prune or delete existing backups as part of the operation.
4. **Report what actually happened**, including dependent rows removed.

**NEVER run against production:**
- `prisma migrate reset` â€” drops and recreates the schema
- `prisma db push --force-reset`
- `prisma db seed` â€” `server/prisma/seed.js` begins by wiping every table
- Any bare `deleteMany` whose cascade you have not counted

`prisma migrate deploy` is the only Prisma command that is safe on production;
it is what the deploy pipeline runs.

**Do not reseed production to make row counts look healthier.** An empty
production database is a fact, not a bug to paper over.

**One named operator at a time.** Two agents working the same repo have already
collided three times: duplicated encrypted-backup implementations, a clobbered
`scripts/backup-crypto.js`, and simultaneous edits to the same incident
document. Before touching shared infrastructure, check whether it already
exists and who owns it.

**Current ownership:** Codex owns backup, encryption and recovery scripts
(`scripts/backup-crypto.js`, `backup-encrypted.js`, `restore-encrypted-backup.js`,
`verify-local-backup.js`, `verify-restore.js`). Do not edit those without
coordinating.
